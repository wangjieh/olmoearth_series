from pathlib import Path
from collections import defaultdict
from PIL import Image
from tqdm import tqdm

# 配置
ORIG_ROOT = Path("./val")           # 原始大图目录
SLICE_ROOT = Path("./val_sliced")   # 切片目录
OUTPUT_ROOT = Path("./val_merged")  # 拼回后的输出目录
SUFFIX = "png"

for folder in ["A", "B", "label"]:
    slice_dir = SLICE_ROOT / folder
    orig_dir = ORIG_ROOT / folder
    out_dir = OUTPUT_ROOT / folder
    out_dir.mkdir(parents=True, exist_ok=True)

    groups = defaultdict(list)

    # 按原图名分组：xxx_0_0.png -> xxx
    for patch_path in slice_dir.glob(f"*.{SUFFIX}"):
        name, x, y = patch_path.stem.rsplit("_", 2)
        groups[name].append((patch_path, int(x), int(y)))

    for name, patches in tqdm(groups.items(), desc=folder):
        orig_path = orig_dir / f"{name}.{SUFFIX}"

        with Image.open(orig_path) as orig_img:
            orig_w, orig_h = orig_img.size

        # 根据切片坐标推断拼接画布大小
        max_w = max(x + Image.open(p).size[0] for p, x, y in patches)
        max_h = max(y + Image.open(p).size[1] for p, x, y in patches)

        with Image.open(patches[0][0]) as first_patch:
            merged = Image.new(first_patch.mode, (max_w, max_h))

            if first_patch.mode == "P" and first_patch.getpalette():
                merged.putpalette(first_patch.getpalette())

        # 粘贴所有 patch
        for patch_path, x, y in patches:
            with Image.open(patch_path) as patch:
                merged.paste(patch, (x, y))

        # 裁回原图尺寸
        merged = merged.crop((0, 0, orig_w, orig_h))

        save_path = out_dir / f"{name}.{SUFFIX}"
        merged.save(save_path)
