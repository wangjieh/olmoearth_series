from pathlib import Path
from PIL import Image
from tqdm import tqdm

src_dir = Path("input_images")
dst_dir = Path("output_images")
dst_dir.mkdir(parents=True, exist_ok=True)

exts = {".jpg", ".jpeg", ".png", ".bmp", ".tif", ".tiff", ".webp"}

img_paths = [p for p in src_dir.rglob("*") if p.suffix.lower() in exts]

for p in tqdm(img_paths, desc="Upscaling images"):
    with Image.open(p) as img:
        w, h = img.size
        out = img.resize((w * 4, h * 4), Image.Resampling.NEAREST)

        save_path = dst_dir / p.relative_to(src_dir)
        save_path.parent.mkdir(parents=True, exist_ok=True)
        out.save(save_path)
